create table users (id INT NOT NULL auto_increment,
					u_name varchar(255),
                    u_password varchar(255),
                    primary key (id)
                    );


create table companies(c_id int primary key not null,
						c_symbol varchar(255),
                        c_name varchar(255),
                        marketcap bigint
                        );



create table watchlist (c_name varchar(255) not null,
						u_id int not null,
                        share_price int,
                        c_symbol varchar(255),
                        foreign key (u_id) references users(id) on delete cascade
                        );


create table portfolio (u_id int,
						c_symbol varchar(255),
                        quantity int,
                        bought_price int,
                        foreign key (u_id) references users(id) on delete cascade
                        );
